## 2. ✍️ `src/data/fetch-tasks.ts`: validated fetch with a signal

```ts
// Fetch and validate tasks from /db.json. Replaced by the Axios API layer in S13.
import type { Task } from '../domain/types';
import { isTask } from '../domain/guards';

export async function fetchTasks(signal?: AbortSignal): Promise<Task[]> {
  const response = await fetch('/db.json', { signal });
  if (!response.ok) {
    throw new Error(`Failed to load tasks: HTTP ${response.status}`);
  }
  const data: unknown = await response.json();
  if (typeof data === 'object' && data !== null && 'tasks' in data && Array.isArray(data.tasks)) {
    const items: unknown[] = data.tasks;
    if (items.every(isTask)) {
      return items;
    }
  }
  throw new Error('Invalid response: expected { tasks: Task[] }');
}
```

This is your S06 `loadTasks` plus one thing: an optional **`AbortSignal`**.

### `AbortController` in one paragraph

```ts
const controller = new AbortController();
fetch(url, { signal: controller.signal });
controller.abort();   // the fetch promise rejects with an AbortError, and the browser cancels the request
```

The controller is the "remote control"; the signal is the "receiver" you hand to `fetch`. Axios accepts the same `signal` (S13).

## 3. ✍️ `src/hooks/useTasks.ts` (first version, with `useState`)

```ts
import { useEffect, useState } from 'react';
import type { Task } from '../domain/types';
import type { RequestState } from '../domain/api-types';
import { fetchTasks } from '../data/fetch-tasks';
import { toErrorMessage } from '../domain/guards';
import { toggleTask } from '../domain/task-utils';

export function useTasks() {
  const [state, setState] = useState<RequestState<Task[]>>({ status: 'idle' });

  useEffect(() => {
    const controller = new AbortController();
    setState({ status: 'loading' });

    fetchTasks(controller.signal)
      .then((tasks) => setState({ status: 'succeeded', data: tasks }))
      .catch((error: unknown) => {
        if (controller.signal.aborted) return; // cancelled on purpose: not an error
        setState({ status: 'failed', error: toErrorMessage(error) });
      });

    return () => controller.abort(); // cleanup: runs before re-running and on unmount
  }, []);

  function toggle(id: number) {
    setState((s) => (s.status === 'succeeded' ? { status: 'succeeded', data: toggleTask(s.data, id) } : s));
  }

  return { state, toggle };
}
```

### Line by line

- **`RequestState<Task[]>`** (06.07): the discriminated union makes "loading and has data" impossible and forces the UI to handle every state.
- **The effect synchronises** "this component is mounted" with "a request for tasks is in flight".
- **`.then/.catch` instead of `async` directly as the effect function**: the effect function must return **nothing or a cleanup function**. An `async` function returns a Promise, which React would treat as an invalid cleanup. Use `.then`, or define an inner `async function load()` and call it.
- **`if (controller.signal.aborted) return;`**: when the cleanup aborts the request, `fetch` rejects with an `AbortError`. That's not a failure the user should see.
- **Cleanup = `controller.abort()`**: if the component unmounts (or StrictMode remounts it), the in-flight request is cancelled and its result can't update state afterwards.
- **`toggle` uses a functional update** and narrows on `status`: you can only toggle when tasks are loaded.

## 4. ✍️ Use it in `App.tsx`

```tsx
import { useState } from 'react';
import type { TaskStatus } from './domain/types';
import { filterTasks } from './domain/task-utils';
import { assertNever } from './domain/guards';
import { useTasks } from './hooks/useTasks';
import { PageLayout } from './components/PageLayout';
import { FilterBar } from './components/FilterBar';
import { TaskList } from './components/TaskList';

export function App() {
  const { state, toggle } = useTasks();
  const [statusFilter, setStatusFilter] = useState<TaskStatus | null>(null);

  let content;
  switch (state.status) {
    case 'idle':
    case 'loading':
      content = <p className="text-muted">Loading tasks…</p>;
      break;
    case 'failed':
      content = <p className="text-danger" role="alert">{state.error}</p>;
      break;
    case 'succeeded': {
      const visible = filterTasks(state.data, { status: statusFilter ?? undefined });
      content = (
        <>
          <FilterBar value={statusFilter} onChange={setStatusFilter} />
          <TaskList tasks={visible} onToggle={toggle} />
        </>
      );
      break;
    }
    default:
      content = assertNever(state);
  }

  return <PageLayout title="Tasks">{content}</PageLayout>;
}
```

## 7. Summary

- A custom hook (`useTasks`) packages state + effect + actions; components just use it.
- `AbortController` + cleanup cancels in-flight requests on unmount/re-run.
- Ignore errors caused by your own abort.
- `RequestState` drives an exhaustive `switch` in the UI.
