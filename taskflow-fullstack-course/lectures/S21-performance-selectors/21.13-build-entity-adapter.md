## 1. ✍️ `src/features/tasks/tasksSlice.ts`

```ts
// S21: the tasks slice on an ENTITY ADAPTER (21.12). Same state shape as S19–S20 ({ ids, entities } + status
// fields), but the CRUD updates and the basic selectors are generated.
import { createEntityAdapter, createSlice, isAnyOf } from '@reduxjs/toolkit';
import type { PayloadAction } from '@reduxjs/toolkit';
import type { Task } from '../../domain/types';
import type { RootState } from '../../app/rootReducer';
import { loggedOut } from '../auth/authActions';
import {
  clearCompleted,
  deleteTask,
  fetchTasks,
  saveNewTask,
  saveTask,
  saveTaskChanges,
  toggleTaskOnServer,
} from './tasksThunks';

export type LoadStatus = 'idle' | 'loading' | 'succeeded' | 'failed';

/**
 * No `sortComparer`: `ids` keep the SERVER's order, and the list's sort preference is applied by a
 * selector (21.05). A sortComparer would re-sort `ids` on every insert and update instead.
 */
const tasksAdapter = createEntityAdapter<Task>();

export interface TasksState extends ReturnType<typeof tasksAdapter.getInitialState> {
  status: LoadStatus;
  error: string | null;
  /** RTK's `meta.requestId` of the fetch in flight; results of any other fetch are ignored (18.11). */
  currentRequestId: string | null;
  /** When the list was last loaded successfully (epoch ms), for "fetch if needed" (18.09). */
  fetchedAt: number | null;
}

export const initialTasksState: TasksState = tasksAdapter.getInitialState({
  status: 'idle' as LoadStatus,
  error: null,
  currentRequestId: null,
  fetchedAt: null,
});

const tasksSlice = createSlice({
  name: 'tasks',
  initialState: initialTasksState,
  reducers: {
    /** A local event kept from S15 (the server marked all tasks done). `prepare` supplies the timestamp. */
    allCompleted: {
      reducer(state, action: PayloadAction<{ updatedAt: string }>) {
        const open = state.ids.filter((id) => state.entities[id]?.status !== 'DONE');
        tasksAdapter.updateMany(
          state,
          open.map((id) => ({ id, changes: { status: 'DONE', updatedAt: action.payload.updatedAt } })),
        );
      },
      prepare(updatedAt: string = new Date().toISOString()) {
        return { payload: { updatedAt } };
      },
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(fetchTasks.pending, (state, action) => {
        state.status = 'loading';
        state.error = null;
        state.currentRequestId = action.meta.requestId;
      })
      .addCase(fetchTasks.fulfilled, (state, action) => {
        if (action.meta.requestId !== state.currentRequestId) return; // stale: ignore (18.10)
        tasksAdapter.setAll(state, action.payload.tasks); // replaces ids AND entities, in the server's order
        state.status = 'succeeded';
        state.currentRequestId = null;
        state.fetchedAt = action.payload.fetchedAt;
      })
      .addCase(fetchTasks.rejected, (state, action) => {
        if (action.meta.requestId !== state.currentRequestId) return;
        state.currentRequestId = null;
        if (action.meta.aborted || action.payload?.kind === 'cancelled') {
          state.status = state.fetchedAt === null ? 'idle' : 'succeeded';
          return;
        }
        state.status = 'failed';
        state.error = action.payload?.message ?? action.error.message ?? 'Unexpected error';
      })
      .addCase(deleteTask.fulfilled, (state, action) => {
        tasksAdapter.removeOne(state, action.payload);
      })
      .addCase(clearCompleted.fulfilled, (state, action) => {
        tasksAdapter.removeMany(state, action.payload);
      })
      .addCase(loggedOut, () => initialTasksState)
      // The server returns the WHOLE task: setOne replaces it (upsertOne would shallow-MERGE into the old one).
      .addMatcher(
        isAnyOf(saveNewTask.fulfilled, saveTask.fulfilled, saveTaskChanges.fulfilled, toggleTaskOnServer.fulfilled),
        (state, action) => {
          tasksAdapter.setOne(state, action.payload);
        },
      );
  },
});

export const { allCompleted } = tasksSlice.actions;
export const tasksReducer = tasksSlice.reducer;

// ── Selectors ─────────────────────────────────────────────────────────────────
// Generated by the adapter, bound to `state.tasks`. selectAll is memoised (same array until ids/entities change).
const adapterSelectors = tasksAdapter.getSelectors((state: RootState) => state.tasks);

export const selectTaskIds = adapterSelectors.selectIds;
export const selectTasks = adapterSelectors.selectAll;
export const selectTaskById = adapterSelectors.selectById;
export const selectTaskCount = adapterSelectors.selectTotal;

export const selectTasksStatus = (state: RootState): LoadStatus => state.tasks.status;
export const selectTasksError = (state: RootState): string | null => state.tasks.error;

/** Returns a number (a primitive), so `useSelector` can compare it with === safely (17.04). */
export const selectOpenTaskCount = (state: RootState): number =>
  selectTasks(state).filter((task) => task.status !== 'DONE').length;
```

## 2. What changed, line by line

| S20 | S21 | Why |
|---|---|---|
| `interface TasksState extends Normalized<Task>` | `extends ReturnType<typeof tasksAdapter.getInitialState>` | the adapter's `EntityState<Task, number>`: the same `{ ids, entities }` |
| a literal initial state | `tasksAdapter.getInitialState({ status, error, … })` | extra fields passed in |
| `normalize` + assigning `ids` and `entities` | `tasksAdapter.setAll(state, tasks)` | the server's full list, in its order |
| `upsert(state, task)` helper | `tasksAdapter.setOne(state, task)` | the server returns the **complete** task: replace, don't merge (21.12 §3) |
| `removeMany(state, ids)` helper | `tasksAdapter.removeOne` / `removeMany` | |
| a loop assigning `status` and `updatedAt` | `tasksAdapter.updateMany(state, open.map(…))` | changes to entities that exist |
| `selectTasks = createSelector(… denormalize …)` | `adapterSelectors.selectAll` | memoized by the adapter |
| `selectTaskById = (state, id) => state.tasks.entities[id]` | `adapterSelectors.selectById` | same behaviour |
| — | `selectTaskCount` (`selectTotal`) | used by the dashboard (21.17) |

The **exported names stay the same** (`selectTasks`, `selectTaskById`, `selectTaskIds`), so no component or selector outside the slice changed. That's the payoff of accessing state only through selectors (21.06).

`normalize.ts` stays: the categories slice still uses it (RTK Query replaces that slice in S22).
