## 1. ✍️ `src/domain.ts`: the TaskFlow vocabulary

These types mirror the backend's Java model ([02-project-spec.md](../../02-project-spec.md) §2). They'll move into the React app in S07 and be refined into API DTOs in S06.

```ts
// TaskFlow domain types (mirror the backend's Java model; see 02-project-spec.md §2)

export type TaskStatus = 'TODO' | 'IN_PROGRESS' | 'DONE';
export type Priority = 'LOW' | 'MEDIUM' | 'HIGH';
export type Role = 'USER' | 'ADMIN';

/** ISO-8601 calendar date, e.g. '2026-10-03'. A type alias documents intent; it doesn't validate. */
export type IsoDate = string;

export interface Category {
  id: number;
  name: string;
  color: string;
}

export interface User {
  id: number;
  username: string;
  displayName: string;
  role: Role;
  locale: 'en' | 'vi';
}

export interface Task {
  id: number;
  title: string;
  description: string;
  status: TaskStatus;
  priority: Priority;
  dueDate: IsoDate | null;
  categoryId?: number;
  ownerId?: number;
}

export interface Comment {
  readonly id: number;
  taskId: number;
  authorId: number;
  body: string;
  createdAt: string;
}

export interface TaskFilter {
  status?: TaskStatus;
  priority?: Priority;
  q?: string;
}

export type SortKey = 'title' | 'priority' | 'dueDate';
export type SortDirection = 'asc' | 'desc';
```

Design decisions:
- **`IsoDate = string`** is only an alias. It documents intent, but TS treats it as any `string`. (S06 mentions "branded types" for stricter modelling; we won't need them.)
- **`dueDate: IsoDate | null`**: always present, may be null (matches the API contract).
- **`categoryId?`, `ownerId?`**: optional *for now*, because the playground data doesn't have them yet. In S06 we separate **what the API returns** (`TaskDto`) from **what a form sends** (`CreateTaskRequest`), and optionality becomes precise.
- **`User` has no `passwordHash`**. The frontend must never receive it (S46). Types are a good place to make that visible.

## 2. ✍️ `src/task-utils.ts`: the S04 functions, typed

Copy `task-utils.js` from `playground/js`, rename it to `.ts`, and add types:

```ts
import type { Priority, SortDirection, SortKey, Task, TaskFilter, TaskStatus } from './domain';

export function addTask(tasks: readonly Task[], task: Task): Task[] {
  return [...tasks, task];
}

export function removeTask(tasks: readonly Task[], id: number): Task[] {
  return tasks.filter((task) => task.id !== id);
}

export function toggleTask(tasks: readonly Task[], id: number): Task[] {
  return tasks.map((task) =>
    task.id === id ? { ...task, status: task.status === 'DONE' ? 'TODO' : 'DONE' } : task,
  );
}

export function filterTasks(tasks: readonly Task[], { status, priority, q }: TaskFilter = {}): Task[] {
  const query = q?.trim().toLowerCase();
  return tasks.filter(
    (task) =>
      (!status || task.status === status) &&
      (!priority || task.priority === priority) &&
      (!query || task.title.toLowerCase().includes(query)),
  );
}

const PRIORITY_RANK: Record<Priority, number> = { LOW: 1, MEDIUM: 2, HIGH: 3 };

export function sortTasks(
  tasks: readonly Task[],
  key: SortKey,
  direction: SortDirection = 'asc',
): Task[] {
  const factor = direction === 'desc' ? -1 : 1;
  return tasks.toSorted((a, b) => {
    let result: number;
    if (key === 'priority') {
      result = PRIORITY_RANK[a.priority] - PRIORITY_RANK[b.priority];
    } else if (key === 'dueDate') {
      if (a.dueDate === b.dueDate) return 0;
      if (a.dueDate === null) return 1;
      if (b.dueDate === null) return -1;
      result = a.dueDate.localeCompare(b.dueDate);
    } else {
      result = a.title.localeCompare(b.title);
    }
    return result * factor;
  });
}

/** ⚠️ First attempt: compiles, but accepts nonsense like ('status', 42). Fixed with generics in 06.02. */
export function updateTaskField(
  tasks: readonly Task[],
  id: number,
  field: keyof Task,
  value: Task[keyof Task],
): Task[] {
  return tasks.map((task) => (task.id === id ? { ...task, [field]: value } : task));
}

export function groupByStatus(tasks: readonly Task[]): Record<TaskStatus, Task[]> {
  const groups: Record<TaskStatus, Task[]> = { TODO: [], IN_PROGRESS: [], DONE: [] };
  for (const task of tasks) {
    groups[task.status].push(task);
  }
  return groups;
}
```

### What the types bought us

1. **`readonly Task[]` parameters**: calling `tasks.sort()` inside any of these is now a **compile error**, not just a frozen-object runtime error.
2. **`Record<Priority, number>`**: forget `HIGH` in the rank table and it won't compile (`Property 'HIGH' is missing…`).
3. **Narrowing in the comparator**: after `if (a.dueDate === null) return 1;`, TS knows `a.dueDate` is `string` in `a.dueDate.localeCompare(…)`.
4. **Literal preservation in `toggleTask`**: the ternary's result `'TODO' | 'DONE'` is assignable to `TaskStatus`, so the returned object is a valid `Task`.
5. **`groups[task.status]`**: because `task.status` is `TaskStatus` and `groups` is a `Record` over exactly those keys, the lookup is known to exist (no `undefined`), even with `noUncheckedIndexedAccess`.

### The weakness in `updateTaskField`

`keyof Task` is `'id' | 'title' | … ` and `Task[keyof Task]` is the union of **all** property types (`number | string | TaskStatus | …`). Nothing links *this* field to *its* type:

```ts
updateTaskField(tasks, 1, 'status', 42);   // ✅ compiles, but it's nonsense!
```

TypeScript accepted it because `42` is in the union (it's a valid `id`). Fixing it needs a **generic** that ties the key to its value type: `<K extends keyof Task>(field: K, value: Task[K])`. That's the first build in S06.

## 3. 📦 Provided: `src/task-utils.test.ts`

```ts
// 📦 Provided tests for task-utils.ts (run: npm test; type-check: npm run typecheck)
import { describe, it, expect } from 'vitest';
import type { Task } from './domain';
import {
  addTask,
  removeTask,
  toggleTask,
  filterTasks,
  sortTasks,
  updateTaskField,
  groupByStatus,
  isOverdue,
} from './task-utils';

function deepFreeze<T>(value: T): T {
  if (value && typeof value === 'object' && !Object.isFrozen(value)) {
    Object.freeze(value);
    Object.values(value).forEach(deepFreeze);
  }
  return value;
}

const task = (overrides: Partial<Task> & Pick<Task, 'id' | 'title'>): Task => ({
  description: '',
  status: 'TODO',
  priority: 'MEDIUM',
  dueDate: null,
  ...overrides,
});

const makeTasks = (): readonly Task[] =>
  deepFreeze([
    task({ id: 1, title: 'Write report', status: 'TODO', priority: 'HIGH', dueDate: '2026-10-03' }),
    task({ id: 2, title: 'fix login bug', status: 'IN_PROGRESS', priority: 'MEDIUM' }),
    task({ id: 3, title: 'Plan offsite', status: 'DONE', priority: 'LOW', dueDate: '2026-09-30' }),
  ]);

const ids = (list: readonly Task[]) => list.map((t) => t.id);

describe('addTask / removeTask / toggleTask', () => {
  it('adds immutably', () => {
    const tasks = makeTasks();
    const next = addTask(tasks, task({ id: 4, title: 'New' }));
    expect(ids(next)).toEqual([1, 2, 3, 4]);
    expect(next).not.toBe(tasks);
  });

  it('removes immutably', () => {
    const tasks = makeTasks();
    expect(ids(removeTask(tasks, 2))).toEqual([1, 3]);
    expect(tasks).toHaveLength(3);
  });

  it('toggles and preserves unchanged references', () => {
    const tasks = makeTasks();
    const next = toggleTask(tasks, 3);
    expect(next[2]?.status).toBe('TODO');
    expect(next[0]).toBe(tasks[0]);
  });
});

describe('filterTasks / sortTasks', () => {
  it('filters', () => {
    const tasks = makeTasks();
    expect(ids(filterTasks(tasks, { status: 'DONE' }))).toEqual([3]);
    expect(ids(filterTasks(tasks, { q: '  LOGIN ' }))).toEqual([2]);
    expect(filterTasks(tasks)).toHaveLength(3);
  });

  it('sorts', () => {
    const tasks = makeTasks();
    expect(ids(sortTasks(tasks, 'priority', 'desc'))).toEqual([1, 2, 3]);
    expect(ids(sortTasks(tasks, 'dueDate'))).toEqual([3, 1, 2]);
  });
});

describe('updateTaskField / groupByStatus', () => {
  it('updates one field immutably', () => {
    const tasks = makeTasks();
    const next = updateTaskField(tasks, 1, 'title', 'Write Q3 report');
    expect(next[0]?.title).toBe('Write Q3 report');
    expect(next[1]).toBe(tasks[1]);
  });

  it('groups with all three keys', () => {
    expect(groupByStatus([])).toEqual({ TODO: [], IN_PROGRESS: [], DONE: [] });
    expect(ids(groupByStatus(makeTasks()).DONE)).toEqual([3]);
  });
});

describe('isOverdue (05.11)', () => {
  it('is true only for unfinished tasks with a past due date', () => {
    const today = '2026-10-01';
    expect(isOverdue(task({ id: 1, title: 'a', dueDate: '2026-09-30' }), today)).toBe(true);
    expect(isOverdue(task({ id: 2, title: 'b', dueDate: '2026-10-01' }), today)).toBe(false);
    expect(isOverdue(task({ id: 3, title: 'c', dueDate: null }), today)).toBe(false);
    expect(isOverdue(task({ id: 4, title: 'd', dueDate: '2026-09-01', status: 'DONE' }), today)).toBe(false);
  });
});
```

Things to notice in the provided tests:
- **The `task()` factory**: `Partial<Task> & Pick<Task, 'id' | 'title'>` means "any Task fields, but `id` and `title` are required". Defaults fill the rest. You'll learn `Partial`/`Pick` in 06.03; for now, read it as a recipe.
- **`next[2]?.status`**: with `noUncheckedIndexedAccess`, `next[2]` is `Task | undefined` (the array might be shorter), so `?.` is needed. In a test, a wrong length makes the assertion fail cleanly.
- **`isOverdue`** doesn't exist yet. It's your 05.11 exercise.
